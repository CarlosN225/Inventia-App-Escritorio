72class MovimientoSerializer(serializers.ModelSerializer): 

73 """Movimientos de inventario, con los nombres del DER. 

74 

75 Campos: id, producto, producto_nombre, tipo_movimiento, cantidad, motivo_merma, 

76 motivo, stock_resultante, usuario, usuario_nombre, venta, compra, fecha_movimiento. 

77 

78 - `tipo_movimiento`: entrada | salida | merma | correccion (minúsculas; se acepta 

79 cualquier mayúscula/minúscula al recibirlo). 

80 - `cantidad`: positiva en entrada, salida y merma. En correccion es la DIFERENCIA 

81 con signo (+ sobrante, - faltante), distinta de 0. 

82 - `motivo_merma`: obligatorio solo en merma. Acepta `consumo` y `devolucion` 

83 además de `consumo_propio` y `devolucion_proveedor`. 

84 - `motivo`: nota. Obligatoria en correccion y cuando la merma es "otro". 

85 - Solo lectura: producto_nombre, usuario, usuario_nombre, stock_resultante, 

86 venta, compra y fecha_movimiento. El usuario lo pone el backend desde la sesión. 

87 

88 Por compatibilidad también se reciben los nombres viejos `tipo` y `observaciones`. 

89 """

127 def to_internal_value(self, data): 

128 # Copia editable de lo que manda el frontend, para normalizar sin tocar el original 

129 data = data.copy() 

130 

131 # Nombres viejos -> nombres del DER (solo si no vino el nuevo) 

132 if "tipo_movimiento" not in data and "tipo" in data: 

133 data["tipo_movimiento"] = data["tipo"] 

134 if "motivo" not in data and "observaciones" in data: 

135 data["motivo"] = data["observaciones"] 

136 

137 tipo = data.get("tipo_movimiento") 

138 if isinstance(tipo, str): 

139 data["tipo_movimiento"] = tipo.strip().lower() 

140 

141 motivo_merma = data.get("motivo_merma") 

142 if motivo_merma in ("", None): 

143 data["motivo_merma"] = None 

144 elif isinstance(motivo_merma, str): 

145 data["motivo_merma"] = self.ALIAS_MOTIVO_MERMA.get(motivo_merma, motivo_merma) 

146 

147 return super().to_internal_value(data) 

148 

149 def validate(self, attrs): 

150 tipo = attrs["tipo_movimiento"] 

151 cantidad = attrs["cantidad"] 

152 motivo = (attrs.get("motivo") or "").strip() 

153 motivo_merma = attrs.get("motivo_merma") 

154 

155 if tipo == MovimientoInventario.TIPO_CORRECCION: 

156 if cantidad == 0: 

157 raise serializers.ValidationError( 

158 {"cantidad": "La corrección debe tener una diferencia distinta de 0."} 

159 ) 

160 if not motivo: 

161 raise serializers.ValidationError( 

162 {"motivo": "La corrección requiere un motivo."} 

163 ) 

164 elif cantidad <= 0: 

165 raise serializers.ValidationError( 

166 {"cantidad": "La cantidad debe ser mayor a 0."} 

167 ) 

168 

169 if tipo == MovimientoInventario.TIPO_MERMA: 

170 if not motivo_merma: 

171 raise serializers.ValidationError( 

172 {"motivo_merma": "La merma requiere un motivo."} 

173 ) 

174 if motivo_merma == MovimientoInventario.MOTIVO_MERMA_OTRO and not motivo: 

175 raise serializers.ValidationError( 

176 {"motivo": "Explica la merma en la nota cuando el motivo es \"otro\"."} 

177 ) 

178 else: 

179 # Solo la merma lleva motivo_merma 

180 attrs["motivo_merma"] = None 

181 

182 attrs["motivo"] = motivo or None 

183 return attrs

291class PromocionSerializer(serializers.ModelSerializer): 

292 class Meta: 

293 model = Promocion 

294 fields = ["id", "producto", "tipo_promocion", "valor", "fecha_inicio", "fecha_fin", "activa"] 

295 

296 def validate(self, datos): 

297 # En una edición parcial (ej. solo pausar), completa con lo que ya tenía 

298 actual = self.instance 

299 tipo = datos.get("tipo_promocion", getattr(actual, "tipo_promocion", None)) 

300 valor = datos.get("valor", getattr(actual, "valor", None)) 

301 inicio = datos.get("fecha_inicio", getattr(actual, "fecha_inicio", None)) 

302 fin = datos.get("fecha_fin", getattr(actual, "fecha_fin", None)) 

303 producto = datos.get("producto", getattr(actual, "producto", None)) 

304 

305 if valor is not None and valor <= 0: 

306 raise serializers.ValidationError({"valor": "El descuento debe ser mayor a 0."}) 

307 if tipo == Promocion.TIPO_PORCENTAJE and valor is not None and valor > 90: 

308 raise serializers.ValidationError({"valor": "El descuento no puede pasar de 90%."}) 

309 if tipo == Promocion.TIPO_MONTO and producto and valor is not None and valor >= producto.precio_venta: 

310 raise serializers.ValidationError({"valor": "El descuento debe ser menor al precio de venta."}) 

311 if inicio and fin and fin < inicio: 

312 raise serializers.ValidationError({"fecha_fin": "La fecha de fin debe ser después del inicio."}) 

313 

314 # Varias promociones sí, pero nunca dos activas en las mismas fechas 

315 activa = datos.get("activa", getattr(actual, "activa", True)) 

316 if activa and producto and inicio and fin: 

317 encimada = ( 

318 Promocion.objects.filter( 

319 producto=producto, 

320 activa=True, 

321 fecha_inicio__lte=fin, 

322 fecha_fin__gte=inicio, 

323 ) 

324 .exclude(pk=getattr(actual, "pk", None)) 

325 .first() 

326 ) 

327 if encimada: 

328 valor_texto = ( 

329 f"{encimada.valor.normalize():f}%" 

330 if encimada.tipo_promocion == Promocion.TIPO_PORCENTAJE 

331 else f"${encimada.valor}" 

332 ) 

333 raise serializers.ValidationError({ 

334 "fecha_inicio": ( 

335 f"Ya hay una promoción activa en esas fechas: {valor_texto} " 

336 f"del {encimada.fecha_inicio:%d/%m} al {encimada.fecha_fin:%d/%m}. " 

337 "Cambia las fechas o pausa la otra." 

338 ) 

339 }) 

340 

341 return datos

344def promocion_vigente(producto): 

345 """La promoción activa de hoy (si hay varias, la de mayor valor). 

346 Usa las promociones ya traídas con prefetch_related: no hace otra consulta por producto.""" 

347 hoy = timezone.localdate() 

348 vigentes = [ 

349 p for p in producto.promociones.all() 

350 if p.activa and p.fecha_inicio <= hoy <= p.fecha_fin 

351 ] 

352 if not vigentes: 

353 return None 

354 

355 promo = max(vigentes, key=lambda p: p.valor) 

356 return { 

357 "id": promo.id, 

358 "tipo": "porcentaje" if promo.tipo_promocion == Promocion.TIPO_PORCENTAJE else "monto", 

359 "valor": str(promo.valor), 

360 "fecha_fin": promo.fecha_fin.isoformat(), 

361 }
