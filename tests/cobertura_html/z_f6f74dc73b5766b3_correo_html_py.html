47def plural(n, uno, varios): 

48 return f"{n} {uno if n == 1 else varios}"

51def fecha_corta(fecha): 

52 return f"{fecha.day} {MESES[fecha.month - 1]}"

55def texto_caducidad(dias): 

56 if dias < 0: 

57 return f"Caducó hace {plural(abs(dias), 'día', 'días')}" 

58 if dias == 0: 

59 return "Caduca hoy" 

60 return f"En {plural(dias, 'día', 'días')}"

67def asunto(resumen, motivo): 

68 """El asunto dice lo importante sin tener que abrir el correo.""" 

69 total = resumen["total_productos"] 

70 

71 if total == 0: 

72 return "INVENTIA · Todo en orden" 

73 

74 if motivo == RECORDATORIO: 

75 return f"INVENTIA · Recordatorio: {plural(total, 'producto pendiente', 'productos pendientes')}" 

76 

77 if motivo == NOVEDAD: 

78 nuevos = [i for i in resumen["stock"] + resumen["caducan"] if i["nuevo"]] 

79 if len(nuevos) == 1: 

80 item = nuevos[0] 

81 if "dias" in item: 

82 if item["dias"] < 0: 

83 return f"INVENTIA · {item['nombre']} ya caducó" 

84 if item["dias"] == 0: 

85 return f"INVENTIA · {item['nombre']} caduca hoy" 

86 return f"INVENTIA · {item['nombre']} caduca en {plural(item['dias'], 'día', 'días')}" 

87 if item["agotado"]: 

88 return f"INVENTIA · Se agotó {item['nombre']}" 

89 return f"INVENTIA · {item['nombre']} bajó de su mínimo" 

90 return f"INVENTIA · {plural(len(nuevos), 'novedad', 'novedades')} en tu inventario" 

91 

92 return f"INVENTIA · {plural(total, 'producto necesita', 'productos necesitan')} atención"

95def encabezado(resumen, motivo): 

96 """Título e introducción según por qué se manda el correo.""" 

97 total = resumen["total_productos"] 

98 

99 if total == 0: 

100 return "Todo en orden", "Ningún producto está debajo de su mínimo ni por caducar." 

101 

102 if resumen.get("primero"): 

103 return ( 

104 "Así está tu inventario hoy", 

105 "Es tu primer aviso. De aquí en adelante solo te escribimos cuando haya algo nuevo.", 

106 ) 

107 

108 if motivo == NOVEDAD: 

109 n = resumen["nuevos"] 

110 return ( 

111 f"Hay {plural(n, 'novedad', 'novedades')} en tu inventario", 

112 "Desde el último aviso cambió lo que está marcado como NUEVO.", 

113 ) 

114 

115 if motivo == RECORDATORIO: 

116 return ( 

117 "Tu recordatorio de los lunes", 

118 "Esto sigue pendiente. Te lo recordamos cada lunes mientras no se resuelva.", 

119 ) 

120 

121 return ( 

122 f"{plural(total, 'producto necesita', 'productos necesitan')} atención", 

123 "Este es el estado de tu inventario en este momento.", 

124 )

127def resumen_corto(resumen): 

128 """Una línea para la vista previa del correo (lo que se lee junto al asunto).""" 

129 partes = [] 

130 if resumen["agotados"]: 

131 partes.append(plural(resumen["agotados"], "agotado", "agotados")) 

132 bajos = len(resumen["stock"]) - resumen["agotados"] 

133 if bajos: 

134 partes.append(plural(bajos, "con stock bajo", "con stock bajo")) 

135 if resumen["caducan"]: 

136 partes.append(plural(len(resumen["caducan"]), "por caducar", "por caducar")) 

137 return " · ".join(partes) or "Todo surtido y nada por caducar"

144def armar_texto(resumen, motivo): 

145 titulo, intro = encabezado(resumen, motivo) 

146 lineas = [ 

147 f"INVENTIA · Resumen del {fecha_corta(resumen['fecha'])}" 

148 + (f" · {resumen['negocio']}" if resumen["negocio"] else ""), 

149 "", 

150 titulo, 

151 intro, 

152 ] 

153 

154 def lista(nombre, items, describir): 

155 if not items: 

156 return 

157 lineas.extend(["", f"{nombre} ({len(items)}):"]) 

158 for item in items[:MAX_POR_LISTA]: 

159 marca = " [NUEVO]" if item["nuevo"] else "" 

160 lineas.append(f"- {item['nombre']}{marca}: {describir(item)}") 

161 if len(items) > MAX_POR_LISTA: 

162 lineas.append(f"…y {len(items) - MAX_POR_LISTA} más en INVENTIA") 

163 

164 lista("Se está acabando", resumen["stock"], texto_stock) 

165 lista("Por caducar", resumen["caducan"], lambda i: texto_caducidad(i["dias"])) 

166 

167 lineas.extend(["", "—", texto_pie()]) 

168 return "\n".join(lineas)

244def armar_html(resumen, motivo, logo_url=None): 

245 titulo, intro = encabezado(resumen, motivo) 

246 subtitulo = f"Resumen del {fecha_corta(resumen['fecha'])}" 

247 if resumen["negocio"]: 

248 subtitulo += f" · {escape(resumen['negocio'])}" 

249 

250 con_caducidad = resumen["maneja_caducidad"] 

251 logo = escape(logo_url) if logo_url else f"cid:{CID_LOGO}" 

252 

253 columna_stock = _columna( 

254 "izq" if con_caducidad else "sola", 

255 "Stock bajo", 

256 len(resumen["stock"]), 

257 plural(resumen["agotados"], "agotado", "agotados") if resumen["agotados"] else "ninguno agotado", 

258 AMBAR, 

259 AMBAR_SUAVE, 

260 resumen["stock"], 

261 lambda i: "Agotado" if i["agotado"] else f"{i['stock']} de {i['minimo']}", 

262 lambda i: (ROJO, True) if i["agotado"] else (GRIS_2, False), 

263 "Todo surtido", 

264 ) 

265 

266 if con_caducidad: 

267 columna_caducidad = _columna( 

268 "der", 

269 "Por caducar", 

270 len(resumen["caducan"]), 

271 f"próximos {resumen['dias_aviso']} días", 

272 ROJO, 

273 ROJO_SUAVE, 

274 resumen["caducan"], 

275 lambda i: texto_caducidad(i["dias"]), 

276 lambda i: (ROJO, True) if i["urgente"] else (AMBAR, True), 

277 "Nada por caducar", 

278 )
